# Collapse evidence cases (docs/EVIDENCE.md). Routine builds use representative public-API cases; the full
# comparison matrix is available with SUB0PUB_COLLAPSE_PROFILE=full. Final-link budgets are checked separately.

set(SUB0PUB_COLLAPSE_PROFILE "smoke" CACHE STRING "Collapse cases to build: smoke or full")
set_property(CACHE SUB0PUB_COLLAPSE_PROFILE PROPERTY STRINGS smoke full)
if(NOT SUB0PUB_COLLAPSE_PROFILE STREQUAL "smoke" AND NOT SUB0PUB_COLLAPSE_PROFILE STREQUAL "full")
    message(FATAL_ERROR "SUB0PUB_COLLAPSE_PROFILE must be smoke or full")
endif()
if(SUB0PUB_COLLAPSE_PROFILE STREQUAL "smoke")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/smoke_cases.txt")
    # One case name per line; blank lines and # comments are skipped, anything else is an error (as in Python)
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/smoke_cases.txt" smoke_lines)
    set(smoke_cases)
    foreach(line ${smoke_lines})
        string(STRIP "${line}" line)
        if(line STREQUAL "" OR line MATCHES "^#")
            continue()
        endif()
        if(NOT line MATCHES "^[a-z_]+$")
            message(FATAL_ERROR "Malformed collapse smoke case line: '${line}'")
        endif()
        list(APPEND smoke_cases "${line}")
    endforeach()
    foreach(smoke_case ${smoke_cases})
        if(NOT IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/cases/${smoke_case}")
            message(FATAL_ERROR "Unknown collapse smoke case: ${smoke_case}")
        endif()
    endforeach()
endif()

file(GLOB SUB0PUB_COLLAPSE_CASES LIST_DIRECTORIES true "${CMAKE_CURRENT_SOURCE_DIR}/cases/*")
foreach(case_dir ${SUB0PUB_COLLAPSE_CASES})
    if(NOT IS_DIRECTORY "${case_dir}")
        continue()
    endif()
    get_filename_component(case "${case_dir}" NAME)
    if(SUB0PUB_COLLAPSE_PROFILE STREQUAL "smoke" AND NOT case IN_LIST smoke_cases)
        continue()
    endif()
    # A variant is <variant>.cpp or a directory <variant>/ of translation units
    file(GLOB entries LIST_DIRECTORIES true "${case_dir}/*")
    set(variants)
    foreach(entry ${entries})
        get_filename_component(name "${entry}" NAME_WE)
        if(IS_DIRECTORY "${entry}" OR entry MATCHES "\\.cpp$")
            list(APPEND variants ${name})
        endif()
    endforeach()
    foreach(form_value 0 1)
        if(form_value)
            set(form observable)
        else()
            set(form removable)
        endif()
        foreach(variant ${variants})
            if(IS_DIRECTORY "${case_dir}/${variant}")
                file(GLOB sources "${case_dir}/${variant}/*.cpp")
            else()
                set(sources "${case_dir}/${variant}.cpp")
            endif()
            set(target Collapse_${case}_${form}_${variant})
            add_executable(${target} driver.cpp ${sources})
            target_link_libraries(${target} PRIVATE Sub0Pub::Sub0Pub)
            target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
            target_compile_definitions(${target} PRIVATE COLLAPSE_OBSERVABLE=${form_value})
            if(NOT variant STREQUAL "handwritten")
                add_test(NAME Collapse_${case}_${form}_${variant}
                         COMMAND ${CMAKE_COMMAND} -DREFERENCE=$<TARGET_FILE:Collapse_${case}_${form}_handwritten>
                                 -DVARIANT=$<TARGET_FILE:${target}>
                                 -P "${CMAKE_CURRENT_SOURCE_DIR}/compare_output.cmake")
            endif()
        endforeach()
    endforeach()
endforeach()

# calls, PLT stubs, MSVC map and dumpbin output), and its gates. The full evidence run is not part of ctest (it
# needs valgrind and the Arm toolchain).

    add_test(NAME Sub0Pub_CollapseEvidenceGates

    add_test(NAME Sub0Pub_CollapseEvidenceSelfTest
             COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/collapse_evidence.py" --self-test)
endif()
